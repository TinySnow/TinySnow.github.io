+make_anchor() {
+  # 为每个文件生成稳定且可跳转的锚点 id。

+  # 1) 先把路径做 slug 化（仅保留字母数字，其他字符压成 -）

+  # sed 正则说明：
+  # - s/[^[:alnum:]]+/-/g: 将连续非字母数字字符压缩为一个 -

+  slug="$(printf '%s' "${file_path}" \

+  if [[ -z "${slug}" ]]; then

+  fi
+

+}

+  deleted="$(printf '%s' "${line}" | awk '{print $2}')"

+    added="0"
+  fi
+  if [[ -z "${deleted}" ]]; then

+  fi

+  printf '%s\t%s\n' "${added}" "${deleted}"

+def iter_name_status_z(output: bytes):
